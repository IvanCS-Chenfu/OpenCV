cmake_minimum_required(VERSION 3.16)
project(opencv_03_modelo_camara LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
add_executable(opencv_03_modelo_camara main.cpp)
target_include_directories(opencv_03_modelo_camara PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(opencv_03_modelo_camara PRIVATE ${OpenCV_LIBS})
