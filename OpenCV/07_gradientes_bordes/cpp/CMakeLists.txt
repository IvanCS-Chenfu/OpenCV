cmake_minimum_required(VERSION 3.16)
project(opencv_07_gradientes_bordes LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
add_executable(opencv_07_gradientes_bordes main.cpp)
target_include_directories(opencv_07_gradientes_bordes PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(opencv_07_gradientes_bordes PRIVATE ${OpenCV_LIBS})
