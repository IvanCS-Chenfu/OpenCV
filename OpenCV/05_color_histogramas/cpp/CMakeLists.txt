cmake_minimum_required(VERSION 3.16)
project(opencv_05_color_histogramas LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
add_executable(opencv_05_color_histogramas main.cpp)
target_include_directories(opencv_05_color_histogramas PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(opencv_05_color_histogramas PRIVATE ${OpenCV_LIBS})
