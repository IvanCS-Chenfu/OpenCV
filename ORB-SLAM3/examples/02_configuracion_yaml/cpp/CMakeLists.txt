cmake_minimum_required(VERSION 3.16)
project(orbslam3_config_yaml LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(OpenCV REQUIRED)
add_executable(orbslam3_config_yaml main.cpp)
target_include_directories(orbslam3_config_yaml PRIVATE ${OpenCV_INCLUDE_DIRS})
target_link_libraries(orbslam3_config_yaml PRIVATE ${OpenCV_LIBS})
